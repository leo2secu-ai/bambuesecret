<script>
  function showTab(id, btn) {
    document.querySelectorAll(".view").forEach(v => v.classList.remove("active"));
    document.querySelectorAll(".tab").forEach(t => t.classList.remove("active"));
    document.getElementById(id).classList.add("active");
    btn.classList.add("active");
  }

  const ABC = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
  const ROTORS = [
    { w: "EKMFLGDQVZNTOWYHXUSPAIBRCJ", n: 16 },
    { w: "AJDKSIRUXBLHWTMCQGZNPYFVOE", n: 4 },
    { w: "BDFHJLCPRTXVZNYEIWGAKMUSQO", n: 21 }
  ];
  const REF = "YRUHQSLDPXNGOKMIEBFZCWVJAT";
  let pos =, inH = "", outH = "", timer = null, client = null;
  const myId = "OP-" + Math.floor(Math.random() * 9000 + 1000);
  const ROWS = ["QWERTZUIO", "ASDFGHJK", "PYXCVBNML"];

  ROWS.forEach(r => {
    let lr = document.createElement("div"), kr = document.createElement("div");
    lr.className = "row"; kr.className = "row";
    for (let c of r) {
      let l = document.createElement("div"); l.className = "lamp"; l.id = "l-" + c; l.innerText = c; lr.appendChild(l);
      let k = document.createElement("button"); k.className = "key"; k.innerText = c; k.onclick = () => press(c); kr.appendChild(k);
    }
    document.getElementById("lamps").appendChild(lr);
    document.getElementById("keys").appendChild(kr);
  });

  function upd() { for (let i = 0; i < 3; i++) document.getElementById("r" + i).innerText = ABC[pos[i]]; }
  function step(i, d) { pos[i] = (pos[i] + d + 26) % 26; upd(); }

  function adv() {
    let m = (pos[0] === ROTORS[0].n), l = (pos[1] === ROTORS[1].n);
    if (l) { pos[1] = (pos[1] + 1) % 26; pos[2] = (pos[2] + 1) % 26; }
    else if (m) { pos[1] = (pos[1] + 1) % 26; }
    pos[0] = (pos[0] + 1) % 26;
    upd();
  }

  function fwd(c, i) { return (ROTORS[i].w.charCodeAt((c + pos[i]) % 26) - 65 - pos[i] + 26) % 26; }
  function bwd(c, i) { return (ROTORS[i].w.indexOf(String.fromCharCode(((c + pos[i]) % 26) + 65)) - pos[i] + 26) % 26; }

  function press(ch) {
    adv();
    let c = ch.charCodeAt(0) - 65;
    c = fwd(fwd(fwd(c, 0), 1), 2);
    c = REF.charCodeAt(c) - 65;
    c = bwd(bwd(bwd(c, 2), 1), 0);
    let out = String.fromCharCode(c + 65);

    document.querySelectorAll(".lamp").forEach(x => x.classList.remove("lit"));
    let el = document.getElementById("l-" + out);
    if (el) el.classList.add("lit");
    clearTimeout(timer);
    timer = setTimeout(() => { if (el) el.classList.remove("lit"); }, 800);

    if (inH.replace(/ /g, "").length > 0 && inH.replace(/ /g, "").length % 5 === 0) { inH += " "; outH += " "; }
    inH += ch; outH += out;
    document.getElementById("inT").innerText = inH;
    document.getElementById("outT").innerText = outH;
  }

  function clr() { inH = ""; outH = ""; document.getElementById("inT").innerText = ""; document.getElementById("outT").innerText = ""; }
  function cpy() { if (outH) { navigator.clipboard.writeText(outH); alert("Copiado."); } }

  function connect() {
    let dot = document.getElementById("dot"); dot.classList.remove("on");
    if (client) client.end();
    let topic = "enigma/radio/" + (document.getElementById("chan").value.trim() || "default");
    client = mqtt.connect("wss://://hivemq.com");
    client.on("connect", () => { dot.classList.add("on"); client.subscribe(topic); });
    client.on("message", (t, m) => {
      try {
        let d = JSON.parse(m.toString());
        if (d.s === myId) return;
        let box = document.getElementById("inbox");
        if (box.innerText.includes("Esperando")) box.innerHTML = "";
        let div = document.createElement("div"); div.className = "msg";
        div.innerHTML = `<div><small style="color:#a89378">DE ${d.s}:</small><br><b style="color:#ffcf40">${d.t}</b></div><button class="dec" onclick="dec('${d.t}')">DESCIFRAR</button>`;
        box.prepend(div);
      } catch (e) {}
    });
  }

  function snd() {
    if (!outH || !client || !client.connected) return alert("Escribí algo o esperá la luz verde.");
    let topic = "enigma/radio/" + (document.getElementById("chan").value.trim() || "default");
    client.publish(topic, JSON.stringify({ s: myId, t: outH }));
    alert("¡Enviado!");
  }

  function dec(txt) { clr(); for (let ch of txt.replace(/[^A-Z]/g, "")) press(ch); }

  let rawHeading = 0, customOffset = 0;
  function handleOrientation(e) {
    let heading = e.webkitCompassHeading || (360 - e.alpha);
    if (heading != null) {
      rawHeading = Math.round(heading);
      let adjusted = (rawHeading - customOffset + 360) % 360;
      document.getElementById("arrow").style.transform = `rotate(${-adjusted}deg)`;
      document.getElementById("degTxt").innerText = adjusted + "°";
    }
  }
  function startCompass() {
    if (typeof DeviceOrientationEvent !== "undefined" && typeof DeviceOrientationEvent.requestPermission === "function") {
      DeviceOrientationEvent.requestPermission().then(res => {
        if (res === "granted") window.addEventListener("deviceorientation", handleOrientation, true);
      });
    } else {
      window.addEventListener("deviceorientationabsolute", handleOrientation, true);
      window.addEventListener("deviceorientation", handleOrientation, true);
    }
    alert("Sensor de brújula activado. Mové el celular en forma de 8 para calibrar.");
  }
  function setCustomNorth() {
    customOffset = rawHeading;
    document.getElementById("compInfo").innerText = "Desvío guardado: " + customOffset + "° (Este es tu nuevo Norte)";
  }

  connect();
</script>
